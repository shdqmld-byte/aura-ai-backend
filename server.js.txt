// Import libraries
const express = require('express');
const axios = require('axios');
const cors = require('cors');
require('dotenv').config();

// Setup express
const app = express();

// Middleware
app.use(express.json());
app.use(cors());

console.log('🚀 Server starting...');
console.log('Environment:', {
  hasGeminiKey: !!process.env.GEMINI_API_KEY,
  hasCloudinaryName: !!process.env.CLOUDINARY_NAME,
  hasCloudinaryPreset: !!process.env.CLOUDINARY_PRESET
});

// ================================================
// ENDPOINT 1: Health Check (untuk test)
// ================================================
app.get('/health', (req, res) => {
  res.json({ 
    status: 'OK',
    message: 'Server is running!',
    timestamp: new Date().toISOString()
  });
});

// ================================================
// ENDPOINT 2: Send Message ke Gemini
// ================================================
app.post('/api/sendMessage', async (req, res) => {
  try {
    console.log('📨 Receiving message request...');
    
    // Get message dari request
    const { message } = req.body;
    
    // Validate
    if (!message || typeof message !== 'string') {
      console.log('❌ Invalid message');
      return res.status(400).json({ 
        error: 'Message harus berupa text'
      });
    }

    const trimmedMessage = message.trim();
    
    if (trimmedMessage.length === 0) {
      console.log('❌ Message kosong');
      return res.status(400).json({ 
        error: 'Pesan tidak boleh kosong'
      });
    }

    console.log('✅ Message valid:', trimmedMessage.substring(0, 50) + '...');
    console.log('📞 Calling Gemini API...');

    // Call Gemini API
    const response = await axios.post(
      `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash-exp:generateContent?key=${process.env.GEMINI_API_KEY}`,
      {
        contents: [{
          parts: [{ 
            text: trimmedMessage 
          }]
        }]
      },
      {
        headers: { 
          'Content-Type': 'application/json' 
        },
        timeout: 30000
      }
    );

    // Extract response
    const aiResponse = response.data.candidates[0].content.parts[0].text;

    console.log('✅ Got response from Gemini');
    console.log('📤 Sending back to app...');

    // Send response back
    return res.json({ 
      success: true, 
      response: aiResponse 
    });

  } catch (error) {
    console.error('❌ Error:', error.message);
    
    if (error.response?.status === 401) {
      console.error('API Key tidak valid!');
      return res.status(401).json({ 
        error: 'API Key error' 
      });
    }
    
    return res.status(500).json({ 
      error: 'Failed to process message',
      details: error.message
    });
  }
});

// ================================================
// ENDPOINT 3: Upload File
// ================================================
app.post('/api/uploadFile', async (req, res) => {
  try {
    console.log('📤 Receiving file upload request...');
    
    const { fileBase64, fileName } = req.body;

    // Validate
    if (!fileBase64 || !fileName) {
      console.log('❌ Invalid file data');
      return res.status(400).json({ 
        error: 'File dan fileName harus ada'
      });
    }

    console.log('📎 File:', fileName);
    console.log('📞 Uploading to Cloudinary...');

    // Setup FormData
    const FormData = require('form-data');
    const form = new FormData();
    
    const buffer = Buffer.from(fileBase64, 'base64');
    form.append('file', buffer, { filename: fileName });
    form.append('upload_preset', process.env.CLOUDINARY_PRESET);

    // Upload
    const response = await axios.post(
      `https://api.cloudinary.com/v1_1/${process.env.CLOUDINARY_NAME}/auto/upload`,
      form,
      { 
        headers: form.getHeaders(),
        timeout: 60000
      }
    );

    const fileUrl = response.data.secure_url;

    console.log('✅ File uploaded successfully');
    console.log('📤 Sending URL back to app...');

    return res.json({ 
      success: true, 
      url: fileUrl,
      message: 'File uploaded'
    });

  } catch (error) {
    console.error('❌ Upload error:', error.message);
    return res.status(500).json({ 
      error: 'Upload failed',
      details: error.message
    });
  }
});

// ================================================
// START SERVER
// ================================================
const PORT = process.env.PORT || 3000;

app.listen(PORT, () => {
  console.log(`\n✅ Server running on port ${PORT}`);
  console.log(`✅ Visit: http://localhost:${PORT}/health`);
  console.log(`\n📝 Endpoints:`);
  console.log(`   - GET  /health`);
  console.log(`   - POST /api/sendMessage`);
  console.log(`   - POST /api/uploadFile\n`);
});

// Handle errors
process.on('unhandledRejection', (reason, promise) => {
  console.error('❌ Unhandled Rejection:', reason);
});
